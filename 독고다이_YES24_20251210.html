# 펼쳐보기 버튼 클릭
click_expand_if_exists <- function() {
  tryCatch({
    button <- remDr_client$findElement("css selector", "em.txt.txt_open")

  }, error = function(e) NULL)
}

# 중복 제거
dedup_text <- function(text) {
  lines <- str_split(text, "\n|<br>|<br/>|<br />", simplify = TRUE)
  lines <- str_trim(lines)
  lines <- lines[lines != ""]
  paste(unique(lines), collapse = " ")
}

# 필드 정제
sanitize_field <- function(text) {
  if (is.na(text)) return(NA_character_)
  text %>%
    str_replace_all("<br>|<br/>|<br />", " ") %>%
    str_replace_all("[\r\n]+", " ") %>%
    str_replace_all("\"", "'") %>%
    str_trim()
}

# 안전 추출
safe_extract <- function(html, selector) {
  tryCatch({
    node <- html %>% html_node(selector)
    if (is.na(node) || length(node) == 0) return(NA_character_)
    html_text(node, trim = TRUE)
  }, error = function(e) NA_character_)
}

# 저자 여러 개인 경우
extract_authors <- function(book_page) {
  tryCatch({
    authors <- book_page %>% html_nodes("span.gd_auth a") %>% html_text(trim = TRUE)
    if (length(authors) == 0) return(NA_character_)
    paste(authors, collapse = ", ")
  }, error = function(e) NA_character_)
}

# 별점 이상치 제거
clean_rating <- function(raw) {
  tryCatch({
    clean <- gsub(",", "", raw)
    num <- as.numeric(clean)
    if (!is.na(num) && num > 10) NA_character_ else as.character(num)
  }, error = function(e) NA_character_)
}

# 안정적 로딩 감지
wait_until_loaded_safe <- function(selector, timeout = 20) {
  for (i in 1:timeout) {
    try({
      nodes <- remDr_client$findElements("css selector", selector)
      if (length(nodes) > 0) return(TRUE)
    }, silent = TRUE)
    Sys.sleep(1)
  }
  return(FALSE)
}

    is.na(x),

Resonance

- 그러면 왜 대학교재는 점점 줄어들까?

공명 요인(Resonance) 데이터 기반 관찰 구조적 원인 외부 증거(뉴스·보고서·정책 근거)
1. 코로나 이후 디지털 강의 전면화 → 교재 발행 감소의 직접 충격 최근 5년 발행량 감소 PPT·PDF 중심의 온라인 강의 급증 → 종이책 구매 필요성 급락 교육부(2021): 대학 강의자료 86% 디지털(PDF·PPT)
동아일보(2021): “PDF 제공 증가로 교재 판매 급감”
한겨레(2020): “교수들이 PDF 제작·배포 증가”
매일경제(2021): “PPT·PDF가 교재 대체”
2. 전자자료 중심 소비 구조 고착 → 종이 교재 메타데이터 약화 책소개·목차·리뷰 결측률 40~80% 대학·도서관이 전자자료 기반 학습으로 전환 → 종이교재 메타데이터 관리 축소 KERIS(2022): 대학 전자자료 이용률 46.9% 증가
대학도서관 전자자료 예산 72% 차지
3. AI 보조학습 도구 확산 → ‘교재 필요성’ 자체 약화 (2차 가속 요인) 발행 감소 시점과 AI 도입 시기 중첩 AI가 요약·번역·문제풀이 등 교재 기능 상당 부분을 대체 → 신간 발행 동기 약화 조선일보(2023): “AI가 과제·요약 대체” 논란
대학가의 AI 활용 지침·제한 증가
4. 국가 정책 자체가 디지털·AI 기반 학습 체제로 전면 이동 발행 감소 + 전자자료 중심 구조 강화 정책 방향이 애초에 ’종이교재 중심 학습’이 아니라 ’AI·디지털 기반 학습’으로 이동 6차 계획(2019–2023): 디지털교과서·전자학술정보 확장, 대학 전자DB 공동구독 확대
2021·2022 시행계획: AI·클라우드 기반 교수학습 환경 강화
7차 계획(2025–2029): “AI 기반 맞춤형 학습 지원 체계 구축”, “지능형 교육 콘텐츠 개발”

Insight

- Data와 Resonance 분석을 종합하면, 교재 감소는 단순한 시장 축소가 아니라
한국 대학 교육 생태계가 디지털·AI 중심 구조로 재편되는 과정이라는 사실이 드러남

이 변화는 교육부의 제6차(2019~2023), 2021·2022 시행계획, 제7차(2024~2028) 교육정보화 계획에서
강조된 방향성과 일치함

아래 3개의 Insight는 이러한 구조적 전환을 설명함

Insight 1. 교재 감소 → 대학 학습 단위의 디지털 재조합의 진행

  • Data: 최근 5년간 발행량 감소 + 텍스트 결측 증가

  • Resonance: 코로나 이후 PDF·PPT 중심 강의 확산 → 교재의 책 단위 필요성 약화

  • 정책 근거:

    • 제6차 교육정보화 계획(2019–2023):
      • 대학 교육에서 “디지털 교수·학습 자원 확대”, “수업자료 온라인 제공체계 강화”를 명시
    • 2021·2022 시행계획:
      • 원격·혼합수업 확산, 학습자료 디지털 제공 의무 강화
    • 제7차 교육정보화 계획:
      • “개인화·모듈화된 디지털 학습체계 구축” 강조

✔ 즉, 교재는 더 이상 책이라는 고정 단위가 아니라
→ 슬라이드·모듈형 콘텐츠·요약본·문제은행으로 분해되고
→ 교수·학생이 AI 도구와 함께 조합하는 방식으로 변하고 있음

대학 교재 감소는 학습 단위의 해체·재조합이라는 구조적 변화의 결과로 볼 수 있음

Insight 2. 대학 출판·교육자료 시장 → AI 활용 가능한 학습 리소스 공급 체계로 이동

  • Data: 책소개·목차·리뷰 결측률 40~80% → 종이책 중심 메타데이터의 약화

  • Resonance: 대학 출판의 비중이 “책 제작”에서 “디지털 자료 제작·관리”로 이동

  • 정책 근거:

    • 제6차 계획:
      • “고등교육 교수·학습 혁신을 위한 디지털 콘텐츠 강화”
    • 2021·2022 시행계획:
      • 대학에 원격수업 콘텐츠 개발 지원
      • 에듀테크 기반 학습자료 개발 사업 확대
    • 제7차 계획:
      • “AI 기반 교수·학습 체제 구축”
      • “지능형 학습 콘텐츠(AI 코스웨어) 개발·보급”

✔ 앞으로 출판사의 경쟁력은
→ AI가 이해하고 가공할 수 있는 구조화된 학습 리소스를 제공하는가로 결정됨

출판 = 콘텐츠 × 데이터 × AI 활용성이라는 새로운 패러다임이 형성되고 있음

Insight 3. 대학생의 학습 방식 → AI 리터러시 기반으로 재편

  • Data: 종이교재 감소 + PDF 강의자료 중심 + 고가 도서 접근성 문제

  • Resonance: 요약·번역·문제풀이 등 주요 기능을 AI가 대체

  • 정책 근거:

    • 제6차·7차 계획 공통:
      • 디지털 소양을 미래 기초역량으로 규정
      • AI 기반 학습지원 서비스 확대
      • AI 활용 평가체계(CBT·지능형 피드백) 도입
    • 대학은 교육부 지원 아래 AI 도입, AI 기반 학습플랫폼 구축을 확대 중

✔ 따라서 학생 역량의 중심축은
→ “교재를 정독하는 능력”에서
→ AI를 활용·판단·조합하는 능력(AI Literacy) 으로 이동하고 있음

이는 대학 교육의 경쟁력이
콘텐츠 이해력 → AI 활용력으로 넘어가고 있음을 의미함

교재 감소는 학습 방식이 AI 중심 패러다임으로 전환되고 있음

Insight — 한 줄 요약

대학교재 감소는 위기가 아니라,
대학 교육이 AI 기반 지능형 학습 체계로 이동하는 구조적 전환의 신호이며
대학·출판사·학생의 역할이 모두 재정의되는 변곡점이다

Problem

- Insight 분석을 종합하면, 교재 감소는 단순한 출판 감소가 아니라
대학생의 학습 방식이 디지털·AI 중심 구조로 이동하고 있다는 신호임

그러나 학습 방식의 변화 속도와
산업계가 요구하는 AI 기반 실무 역량의 변화 속도는 일치하지 않음

이로 인해 자연스럽게 다음과 같은 문제가 나타나고 있음:

“기업이 요구하는 역량”과
“대학생이 실제로 준비하는 역량” 사이의 격차가 커지고 있음

따라서 문제의 핵심 질문은 단순히

“교재가 줄었는가?”가 아니라,
“기업이 요구하는 AI 기반 역량을 대학생이 충분히 준비하고 있는가?” 임

  • 그러나 대학 수업은 여전히 교재 기반·암기 중심 평가에 머무르는 경우가 많아, 기업이 요구하는 역량과 대학생이 준비하는 역량의 간극이 커지고 있음

Problem 2. 산업계는 AI 협업 역량을 요구하지만, 대학 교육은 이를 충분히 제공하지 못함

✔ 글로벌 보고서 근거

  • Deloitte AI Industry Report (2024)
    → “전 산업의 핵심 경쟁력은 AI-assisted work로 이동 중”이라고 분석함

  • McKinsey Future of Work (2023)
    → “AI와 협업하며 일할 수 있는 역량(AI-augmented skill) 이
    신규 채용에서 가장 중요한 역량 중 하나”라고 제시함

✔ 실제 직무 변화 예시

산업 과거 업무 방식 현재/미래 요구 역량
마케팅 보고서 작성, 자료조사 수작업 AI 기반 시장 분석, 자동 리포트 생성
데이터·개발 수작업 코딩·전처리 Copilot 기반 개발, AI 코드 리뷰, 프롬프트 설계
경영·기획 엑셀·PPT 기반 정적 분석 AI 활용 전략 시뮬레이션, 시나리오 플래닝
교육·출판 종이 교재 제작 중심 AI 코스웨어, 디지털 학습 리소스·플랫폼 설계

➜ 의미

  • 겉으로는 PDF·슬라이드 중심 디지털 강의가 확대되었지만,실제로 산업에서 요구하는 역량은 AI와 협업하는 실무 능력임

  • 그러나 이러한 역량은 현행 대학 교육만으로는 충분히 준비하기 어려운 구조임

Problem 3. AI 확산 - 학생 간 역량 격차 증가

✔ 실제로 나타나는 현상

  • AI 활용이 능숙한 학생: 분석·정리·코딩·문서 작성까지 자동화 → 성과 급상승
  • AI 활용이 미숙한 학생: 모든 작업을 수작업으로 수행 → 속도·결과 모두 뒤처짐

과거에는 모두가 동일한 교재로 비슷한 출발선을 가졌지만,
이제는 AI 활용 능력 자체가 학습 성과·취업 경쟁력을 결정하는 핵심 변수가 됨

즉,

AI를 잘 쓰는 학생과 그렇지 못한 학생의 격차는
교재 시대보다 훨씬 더 빠르고 크게 벌어지고 있음

Problem 4. 앞으로 대학 경쟁력을 결정하는 것은 ‘전공 지식’이 아니라 ‘AI 기반 실무 역량’

✔ 앞으로 대학생이 반드시 갖춰야 할 5대 핵심 역량

  1. AI 활용 능력 (AI Literacy)
    • ChatGPT, Copilot 등 생성형 AI 도구를 활용한 분석·요약·문제 해결
    • AI 결과물의 타당성을 검증하는 비판적 사고력
    • 업무·학습 맥락에 맞는 프롬프트 설계 능력
  2. 디지털 데이터 해석 능력
    • PDF, 리포트, 온라인 자료 등 디지털 텍스트·표·그래프를
      구조적으로 읽고, 필요한 정보를 추출·조합하는 능력
  3. 자동화·생산성 도구 활용 능력
    • 반복 작업을 줄이고,
      AI + 스크립트 + 툴(예: 노션, 파이썬, R, Zapier 등) 을 조합해
      워크플로우를 설계·자동화하는 능력
  4. 프로젝트 기반 문제 해결 능력
    • 이론 암기가 아니라,
      “문제를 정의 → 데이터·AI를 활용해 해결안 설계 → 결과물로 보여주는 능력”
    • 기업은 성적보다 포트폴리오·프로젝트 경험을 더 중요하게 보는 추세임
  5. 디지털 커뮤니케이션 능력
    • AI를 활용한 문서·리포트 작성,
    • 협업툴(Jira, Notion, GitHub 등)을 통한 팀 단위 소통과 기록 관리 능력

Problem — 최종 정리

교재 감소는 단순한 출판 시장의 변화가 아니라,
대학생에게 요구되는 역량의 기준이 바뀌고 있다는 신호임

기업은 이미 AI와 함께 일할 수 있는 실무형 인재를 요구하고 있지만,
많은 대학생은 여전히 교재·시험 중심 학습 구조에 머물러 있음

따라서 지금 필요한 것은 교재의 대체가 아니라,
AI 시대에 생존 가능한 역량을 학생들이 갖출 수 있도록
학습·교육의 방향을 재설정하는 것임
